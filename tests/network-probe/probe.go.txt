package program

import (
	"example.com/probe/libs/bend"
	"example.com/probe/libs/quote"
	"example.com/probe/model"
	"gosvm/solana"
)

// Probe composes Bend and Rust and returns the 8-byte quote as return data,
// so a deployed program can be checked without any state account.
func Probe(c solana.Context, accounts ProbeAccounts, args model.ProbeArgs) (ProbeAccounts, uint64) {
	var request [6]byte
	request[1] = args.Enabled
	for i := uint64(0); i < 4; i++ {
		request[2+i] = byte(args.Value >> (i * 8))
	}
	var decision [5]byte
	status, length := bend.Evaluate(request[:], decision[:])
	if status != 0 || length != 5 {
		return accounts, 100 + uint64(status)
	}
	if decision[0] != 0 {
		return accounts, 200
	}
	var input [16]byte
	for i := uint64(0); i < 8; i++ {
		input[i] = byte(args.Amount >> (i * 8))
	}
	for i := uint64(0); i < 4; i++ {
		input[8+i] = decision[1+i]
		input[12+i] = byte(args.Reserve >> (i * 8))
	}
	var output [8]byte
	status, length = quote.Quote(input[:], output[:])
	if status != 0 || length != 8 {
		return accounts, 300 + uint64(status)
	}
	return accounts, solana.SetReturnData(c, output[:])
}
