func decode(s string)[]byte{b,e:=hex.DecodeString(s);if e!=nil{panic(e)};return b[:len(b):len(b)]}

func main() {
	if len(os.Args)==3&&os.Args[1]=="--key"{fmt.Printf("%x\n",ed25519.NewKeyFromSeed(decode(os.Args[2])).Public().(ed25519.PublicKey));return}
	var rows []row;if e:=json.NewDecoder(os.Stdin).Decode(&rows);e!=nil{panic(e)}
	var results []result
	for _,r:=range rows {
		c:=solana.Context{ID:decode(r.ID),InstructionData:decode(r.Instruction)}
		for _,a:=range r.Accounts{c.Accounts=append(c.Accounts,solana.Account{Key:decode(a.Key),Owner:decode(a.Owner),Data:decode(a.Data),Signer:a.Signer,Writable:a.Writable,Executable:a.Executable,Lamports:a.Lamports})}
		out:=result{Name:r.Name,Data:[]string{},Lamports:[]uint64{}}

			out.Calls++

			}

			s,d,a:=indices[0],indices[1],indices[2];source,destination:=c.Accounts[s].Data,c.Accounts[d].Data
			if source[108]==2||destination[108]==2{return 17}
			if !bytes.Equal(source[:32],destination[:32]){return 3}
			if !bytes.Equal(source[32:64],c.Accounts[a].Key){return 4}
			if !c.Accounts[a].Signer {
				valid:=false;off:=1
				for g:=0;g<int(seeds[0]);g++{start:=off;n:=int(seeds[off]);off++;for j:=0;j<n;j++{size:=int(seeds[off]);off+=1+size};address,code:=derive(seeds[start:off],c.ID);if code==0&&bytes.Equal(address,c.Accounts[a].Key){valid=true}}
				if !valid{panic("invalid fixture signer seeds")}
			}
			amount:=binary.LittleEndian.Uint64(data[1:]);x,y:=binary.LittleEndian.Uint64(source[64:]),binary.LittleEndian.Uint64(destination[64:]);if x<amount{return 1};if ^uint64(0)-y<amount{return 14};binary.LittleEndian.PutUint64(source[64:],x-amount);binary.LittleEndian.PutUint64(destination[64:],y+amount);return 0
		}

		for _,a:=range c.Accounts{out.Data=append(out.Data,hex.EncodeToString(a.Data));out.Lamports=append(out.Lamports,a.Lamports)}
		results=append(results,out)
	}
	if e:=json.NewEncoder(os.Stdout).Encode(results);e!=nil{panic(e)}
}
